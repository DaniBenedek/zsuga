angular.module('app', ['ui.router', 'app.common'])
.config(['$stateProvider', '$urlRouterProvider', function($sp, $url) {
    $sp
    .state('root', {
        views: {
            '': { templateUrl: './html/root.html' },
            'header@root': { templateUrl: './html/header.html' }
        }
    })
    .state('home', {
        url: '/', parent: 'root',
        templateUrl: './html/home.html',
        controller: 'homeController'
    })
    .state('movies', {
        url: '/movies', parent: 'root',
        templateUrl: './html/movies.html',
        controller: 'moviesController'
    });
    $url.otherwise('/');
}]);
